! ===== Core-2 (Layer 3 switch, HSRP STANDBY) - example values =====
! Same as Core-1 except: SVI addresses end in .3 (.131 / .67), default priority 100,
! and uplink to the gateway uses 10.20.2.102/30.
enable
configure terminal
hostname CORE-2
ip routing
!
vlan 10
vlan 20
vlan 30
vlan 40
vlan 50
exit
!
interface vlan 10
 ip address 10.20.0.3 255.255.255.0
 standby 10 ip 10.20.0.1
 no shutdown
interface vlan 20
 ip address 10.20.1.3 255.255.255.128
 standby 20 ip 10.20.1.1
 no shutdown
interface vlan 30
 ip address 10.20.1.131 255.255.255.128
 standby 30 ip 10.20.1.129
 no shutdown
interface vlan 40
 ip address 10.20.2.3 255.255.255.192
 standby 40 ip 10.20.2.1
 no shutdown
interface vlan 50
 ip address 10.20.2.67 255.255.255.224
 standby 50 ip 10.20.2.65
 no shutdown
exit
!
interface range fa0/1 - 5
 switchport trunk encapsulation dot1q
 switchport mode trunk
exit
!
interface fa0/6
 no switchport
 ip address 10.20.2.102 255.255.255.252
 no shutdown
exit
!
router rip
 version 2
 no auto-summary
 network 10.0.0.0
end
write memory
